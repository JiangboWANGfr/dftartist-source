cmake_minimum_required(VERSION 3.20)
project(BInvestigationCatalogConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
if(HIER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(hier_warnings INTERFACE)
target_compile_options(hier_warnings INTERFACE -Wall -Wextra -Wpedantic -Werror)
add_library(hiercore STATIC IMPORTED GLOBAL)
set_target_properties(hiercore PROPERTIES IMPORTED_LOCATION "${B_CORE}/core/libhiercore.a"
 INTERFACE_INCLUDE_DIRECTORIES "${HIER_B_PRODUCT}/core/query/include;${HIER_B_PRODUCT}/core/model/include;${HIER_B_PRODUCT}/core/db/include;${HIER_B_PRODUCT}/core/frontend/include")
target_link_libraries(hiercore INTERFACE "${B_CORE}/core/libhier_db.a" "${B_CORE}/core/libhier_frontend.a" "${B_CORE}/_deps/slang-build/lib/libsvlang.a")
add_library(hier_protocol STATIC IMPORTED GLOBAL)
set_target_properties(hier_protocol PROPERTIES IMPORTED_LOCATION "${B_CORE}/core/libhier_protocol.a"
 INTERFACE_INCLUDE_DIRECTORIES "${HIER_B_PRODUCT}/core/protocol/include")
target_link_libraries(hier_protocol INTERFACE hiercore)
add_subdirectory("${HIER_B_PRODUCT}/analysis" analysis)
add_executable(b_scenarios_contract "${CMAKE_CURRENT_SOURCE_DIR}/../independent-scenarios/unit.cpp")
target_link_libraries(b_scenarios_contract PRIVATE dft_sequential hier_warnings)
add_test(NAME b_scenarios_contract COMMAND b_scenarios_contract "${CMAKE_CURRENT_SOURCE_DIR}/../independent-scenarios/design.sv")
add_executable(b_previous_sequential "${CMAKE_CURRENT_SOURCE_DIR}/../independent-sequential/unit.cpp")
target_link_libraries(b_previous_sequential PRIVATE dft_sequential hier_warnings)
add_test(NAME b_previous_sequential COMMAND b_previous_sequential "${CMAKE_CURRENT_SOURCE_DIR}/../independent-sequential/design.sv")
add_executable(b_scenarios_utf8 "${CMAKE_CURRENT_SOURCE_DIR}/../independent-scenarios/utf8.cpp")
target_link_libraries(b_scenarios_utf8 PRIVATE dft_sequential hier_warnings)
add_test(NAME b_scenarios_utf8 COMMAND b_scenarios_utf8 "${CMAKE_CURRENT_SOURCE_DIR}/../independent-scenarios/design.sv")

add_executable(b_investigation_catalog_contract unit.cpp)
target_link_libraries(b_investigation_catalog_contract PRIVATE dft_sequential dft_objects hier_warnings)
add_test(NAME b_investigation_catalog_contract COMMAND b_investigation_catalog_contract "${CMAKE_CURRENT_SOURCE_DIR}/design.sv" "${CMAKE_CURRENT_SOURCE_DIR}/catalog.sv")
set_tests_properties(b_investigation_catalog_contract PROPERTIES TIMEOUT 60)
