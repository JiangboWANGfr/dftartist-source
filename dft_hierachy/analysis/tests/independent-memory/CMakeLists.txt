cmake_minimum_required(VERSION 3.20)
project(BMemoryConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
if(HIER_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(hier_warnings INTERFACE)
target_compile_options(hier_warnings INTERFACE -Wall -Wextra -Wpedantic -Werror)
add_library(hiercore STATIC IMPORTED GLOBAL)
set_target_properties(hiercore PROPERTIES IMPORTED_LOCATION "${B_CORE}/core/libhiercore.a"
 INTERFACE_INCLUDE_DIRECTORIES "${HIER_B_PRODUCT}/core/query/include;${HIER_B_PRODUCT}/core/model/include;${HIER_B_PRODUCT}/core/db/include;${HIER_B_PRODUCT}/core/frontend/include")
target_link_libraries(hiercore INTERFACE "${B_CORE}/core/libhier_db.a" "${B_CORE}/core/libhier_frontend.a" "${B_CORE}/_deps/slang-build/lib/libsvlang.a")
add_library(hier_protocol STATIC IMPORTED GLOBAL)
set_target_properties(hier_protocol PROPERTIES IMPORTED_LOCATION "${B_CORE}/core/libhier_protocol.a"
 INTERFACE_INCLUDE_DIRECTORIES "${HIER_B_PRODUCT}/core/protocol/include")
target_link_libraries(hier_protocol INTERFACE hiercore)
add_subdirectory("${HIER_B_PRODUCT}/analysis" analysis)
add_executable(b57_memory_contract unit.cpp)
target_link_libraries(b57_memory_contract PRIVATE dft_memory hier_protocol hier_warnings)
add_test(NAME b57_memory_contract COMMAND b57_memory_contract "${CMAKE_CURRENT_SOURCE_DIR}/design.sv")
set_tests_properties(b57_memory_contract PROPERTIES TIMEOUT 120)
