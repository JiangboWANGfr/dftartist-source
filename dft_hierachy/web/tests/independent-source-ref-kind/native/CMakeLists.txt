cmake_minimum_required(VERSION 3.28)
project(b_ref_kind_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT IS_ABSOLUTE "${HIER_PRODUCT}")
  message(FATAL_ERROR "HIER_PRODUCT must be a fixed absolute product checkout")
endif()
set(HIER_BUILD_EXPERIMENTS OFF CACHE BOOL "" FORCE)
set(HIER_TEST_OBSERVER ON CACHE BOOL "" FORCE)
add_subdirectory("${HIER_PRODUCT}/core" "product-core")
add_executable(b_ref_kind_api CoreApi.cpp)
target_link_libraries(b_ref_kind_api PRIVATE hiercore)
if(HIER_SANITIZE)
  target_compile_options(b_ref_kind_api PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  target_link_options(b_ref_kind_api PRIVATE -fsanitize=address,undefined)
endif()
